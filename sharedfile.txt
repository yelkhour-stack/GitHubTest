        .data
sieve:  .space 100          # 100 bytes for indices 0..99

    .text
    .globl main

main:
    ################################################################
    # Constants and base addresses (loaded once)
    ################################################################
    li   t7, 0x1000         # t7 = base address of sieve
    li   s10, 0x10000000    # s10 = output MMIO base
    li   t2, 100            # t2 = 100 (upper bound)
    li   t6, 10             # t6 = 10  (used in several places)

    ################################################################
    # 1) Initialize sieve: set all entries to 0
    ################################################################
    mv   t0, t7             # t0 = &sieve[0]
    li   t1, 0              # i = 0

InitZeroLoop:
    bge  t1, t2, InitOnes   # if i >= 100, stop
    sb   x0, 0(t0)          # sieve[i] = 0
    addi t0, t0, 1          # next address
    addi t1, t1, 1          # i++
    j    InitZeroLoop

InitOnes:
    # Set sieve[i] = 1 for i = 2..99
    addi t0, t7, 2          # t0 = &sieve[2]
    li   t1, 2              # i = 2

InitOnesLoop:
    bge  t1, t2, SieveDoneInit

    li   t3, 1
    sb   t3, 0(t0)          # sieve[i] = 1

    addi t0, t0, 1          # &sieve[i+1]
    addi t1, t1, 1          # i++
    j    InitOnesLoop

SieveDoneInit:
    ################################################################
    # 2) Sieve of Eratosthenes: mark multiples as not prime
    ################################################################
    li   t5, 2              # p = 2

OuterPrimeLoop:
    bge  t5, t6, PrintPrimes  # stop when p >= 10

    # check if sieve[p] == 1
    add  t1, t7, t5         # &sieve[p] = base + p
    lb   t3, 0(t1)          # t3 = sieve[p]
    beq  t3, x0, NextP      # if not prime, skip

    # k = 2p
    add  t3, t5, t5         # t3 = k = 2p

MarkMultiplesLoop:
    bge  t3, t2, NextP      # if k >= 100, break

    add  t1, t7, t3         # &sieve[k]
    sb   x0, 0(t1)          # sieve[k] = 0

    add  t3, t3, t5         # k += p
    j    MarkMultiplesLoop

NextP:
    addi t5, t5, 1          # p++
    j    OuterPrimeLoop

    ################################################################
    # 3) Scan sieve and PRINT ONLY PRIMES (2..99)
    #    No div/mul: keep decimal digits as counters
    ################################################################
PrintPrimes:
    li   s0, 0              # tens digit (0..9)
    li   s1, 0              # ones digit (0..9)
    li   s2, 0              # i = 0..99

    # Advance until i == 2, updating decimal digits
SkipToTwo:
    li   t0, 2
    bge  s2, t0, ScanLoop   # if i >= 2, start scanning

    addi s2, s2, 1          # i++
    addi s1, s1, 1          # ones++

    bne  s1, t6, SkipToTwo  # if ones != 10, loop

    li   s1, 0              # ones = 0
    addi s0, s0, 1          # tens++
    j    SkipToTwo

ScanLoop:
    bge  s2, t2, Done       # if i >= 100, finished

    # check if i is prime: sieve[i] == 1 ?
    add  t1, t7, s2         # &sieve[i]
    lb   t3, 0(t1)
    beq  t3, x0, IncNumber  # not prime, skip print

    # ---- PRINT current prime (two digits) ----
    # print tens digit: '0' + s0
    addi a1, s0, 48         # ASCII '0' = 48
    sb   a1, 0(s10)

    # print ones digit: '0' + s1
    addi a1, s1, 48
    sb   a1, 0(s10)

    # newline
    li   a1, 10             # '\n'
    sb   a1, 0(s10)

IncNumber:
    # increment i and its decimal digits by 1
    addi s2, s2, 1          # i++
    addi s1, s1, 1          # ones++

    bne  s1, t6, ScanLoop   # if ones != 10, continue

    li   s1, 0              # ones = 0
    addi s0, s0, 1          # tens++
    j    ScanLoop

Done:
    ebreak                  # stop execution
